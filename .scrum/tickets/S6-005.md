### 切替ロジック（参考実装）

```swift
struct BatteryAdaptiveLocationPolicy {
    func evaluate(recentLocations: [CLLocation], homeRegion: CLCircularRegion?) -> Decision {
        if isAtHome(latest: recentLocations.last, home: homeRegion) {
            return .stopRecording
        }
        let movedDistance = distance(in: recentLocations.suffix(5)) // 直近 5 件
        if movedDistance > 100 {
            return .driving // best / 10m
        } else {
            return .stopped // hundredMeters / 100m
        }
    }
}
```
